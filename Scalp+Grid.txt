//| SAFE COPY BUFFER                                                  |
//+------------------------------------------------------------------+
bool SafeCopyBuffer(int handle, int bufferNum, int startPos, int count, double &arr[])
  {
   if(handle == INVALID_HANDLE)
      return false;
   ArrayResize(arr, count);
   ArraySetAsSeries(arr, true);
   int copied = CopyBuffer(handle, bufferNum, startPos, count, arr);
   return (copied >= count);
  }

   ArraySetAsSeries(rates, true);
   int copied = CopyRates(_Symbol, _Period, startPos, count, rates);
   return (copied >= count);
  }

//+------------------------------------------------------------------+
//| CALCULATE LSMA (Least Squares Moving Average)                     |
//| Uses proper linear regression with extrapolation                  |
//+------------------------------------------------------------------+
double CalcLSMA(const double &prices[], int startIdx, int period, int arrSize)
  {
   if(startIdx < 0 || period <= 1 || (startIdx + period) > arrSize)
      return 0;

   double sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
   double n = (double)period;

   for(int i = 0; i < period; i++)
     {
      int idx = startIdx + i;
      if(idx >= arrSize)
         return 0;
      double x = (double)(period - 1 - i);  // newest bar = highest x
      double y = prices[idx];
      sumX  += x;
      sumY  += y;
      sumXY += x * y;
      sumX2 += x * x;
     }

   double denom = (n * sumX2 - sumX * sumX);
   if(MathAbs(denom) < 1e-12)
      return sumY / n;  // Flat line fallback

   double slope     = (n * sumXY - sumX * sumY) / denom;
   double intercept = (sumY - slope * sumX) / n;

   // Extrapolate to the most recent point (x = n-1)
   return intercept + slope * (n - 1.0);
  }

//+------------------------------------------------------------------+
//| CALCULATE ZLSMA (Zero Lag LSMA)                                   |
//| Proper implementation: 2 * LSMA(price) - LSMA(LSMA(price))      |
//+------------------------------------------------------------------+
double CalcZLSMA(int shift)
  {
   int period = InpZLSMAPeriod;
   if(period <= 1)
      return 0;

   int needBars = shift + period * 2 + 10;

   MqlRates rates[];
   if(!SafeCopyRates(rates, 0, needBars))
      return 0;

   int ratesSize = ArraySize(rates);
   if(ratesSize < needBars)
      return 0;

   // Extract close prices
   double closes[];
   ArrayResize(closes, ratesSize);
   for(int i = 0; i < ratesSize; i++)
      closes[i] = rates[i].close;

   // Step 1: LSMA of close prices at the target shift
   double lsma1 = CalcLSMA(closes, shift, period, ratesSize);
   if(lsma1 == 0)
      return 0;

   // Step 2: Build array of LSMA values for period bars starting from shift
   double lsmaArr[];
   ArrayResize(lsmaArr, period);
   for(int j = 0; j < period; j++)
     {
      int startIdx = shift + j;
      if(startIdx + period > ratesSize)
         return 0;
      lsmaArr[j] = CalcLSMA(closes, startIdx, period, ratesSize);
      if(lsmaArr[j] == 0)
         return 0;
     }

   // Step 3: LSMA of LSMA values
   double lsma2 = CalcLSMA(lsmaArr, 0, period, period);
   if(lsma2 == 0)
      return 0;

   // Zero-lag correction: 2 * LSMA - LSMA(LSMA)
   return 2.0 * lsma1 - lsma2;
  }

//+------------------------------------------------------------------+
//| BUILD HEIKIN ASHI CACHE                                           |
//| Proper recursive HA calculation                                   |
//+------------------------------------------------------------------+
bool BuildHeikinAshiCache(int count)
  {
   MqlRates rates[];
   // Need extra bars for proper recursive initialization
   int extra = 50;
   int total = count + extra;

   if(!SafeCopyRates(rates, 0, total))
      return false;

   int ratesSize = ArraySize(rates);
   if(ratesSize < total)
      return false;

   ArrayResize(g_haOpen, ratesSize);
   ArrayResize(g_haClose, ratesSize);
   ArrayResize(g_haHigh, ratesSize);
   ArrayResize(g_haLow, ratesSize);

   // Build from oldest to newest (rates are series: [0]=newest)
   // Start from the oldest bar
   int oldest = ratesSize - 1;

   // First HA bar
   g_haClose[oldest] = (rates[oldest].open + rates[oldest].high +
                         rates[oldest].low + rates[oldest].close) / 4.0;
   g_haOpen[oldest]  = (rates[oldest].open + rates[oldest].close) / 2.0;
   g_haHigh[oldest]  = rates[oldest].high;
   g_haLow[oldest]   = rates[oldest].low;

   // Build forward (from oldest-1 toward index 0)
   for(int i = oldest - 1; i >= 0; i--)
     {
      g_haClose[i] = (rates[i].open + rates[i].high +
                       rates[i].low + rates[i].close) / 4.0;
      g_haOpen[i]  = (g_haOpen[i + 1] + g_haClose[i + 1]) / 2.0;
      g_haHigh[i]  = MathMax(rates[i].high, MathMax(g_haOpen[i], g_haClose[i]));
      g_haLow[i]   = MathMin(rates[i].low, MathMin(g_haOpen[i], g_haClose[i]));
     }

   g_haCacheBar = Bars(_Symbol, _Period);
   return true;
  }

//+------------------------------------------------------------------+
//| CALCULATE CHANDELIER EXIT                                         |
//| Proper recursive implementation with full state tracking          |
//+------------------------------------------------------------------+
bool CalcChandelierExit()
  {
   int period = InpCEPeriod;
   if(period <= 0)
      return false;

   int needBars = period * 3 + 10;

   MqlRates rates[];
   if(!SafeCopyRates(rates, 0, needBars))
      return false;

   int ratesSize = ArraySize(rates);
   if(ratesSize < period + 5)
      return false;

   double atrBuf[];
   if(!SafeCopyBuffer(g_hATR_CE, 0, 0, needBars, atrBuf))
      return false;

   int atrSize = ArraySize(atrBuf);
   if(atrSize < 5)
      return false;

   // We need to compute CE for bars going back far enough to establish state,
   // then use the final values. Compute for bars [period+5] down to [1].

   // Initialize from deep history
   int startBar = MathMin(ratesSize - period - 1, atrSize - 1);
   startBar = MathMin(startBar, period * 2);

   if(startBar < 3)
      return false;

   // Calculate initial values at startBar
   double prevLongStop  = 0;
   double prevShortStop = 0;
   int    prevDir       = 0;
   bool   firstCalc     = true;

   for(int bar = startBar; bar >= 1; bar--)
     {
      if(bar >= ratesSize || bar >= atrSize)
         continue;

      // Highest high and lowest low over [bar .. bar+period-1]
      double hh = -DBL_MAX;
      double ll = DBL_MAX;
      for(int k = bar; k < bar + period && k < ratesSize; k++)
        {
         if(rates[k].high > hh) hh = rates[k].high;
         if(rates[k].low < ll)  ll = rates[k].low;
        }

      if(hh == -DBL_MAX || ll == DBL_MAX)
         continue;
      if(bar >= atrSize || atrBuf[bar] <= 0)
         continue;

      double longStop  = hh - InpCEMultiplier * atrBuf[bar];
      double shortStop = ll + InpCEMultiplier * atrBuf[bar];

      if(!firstCalc)
        {
         // Ratchet: only move stops in favorable direction
         // For long stop: if previous close was above previous long stop,
         // new long stop can only go higher
         int prevBar = bar + 1;
         if(prevBar < ratesSize)
           {
            if(rates[prevBar].close > prevLongStop)
               longStop = MathMax(longStop, prevLongStop);
            if(rates[prevBar].close < prevShortStop)
               shortStop = MathMin(shortStop, prevShortStop);
           }

         // Direction determination
         if(rates[bar].close > prevShortStop)
            prevDir = 1;
         else if(rates[bar].close < prevLongStop)
            prevDir = -1;
         // else: keep previous direction
        }
      else
        {
         // First calculation - determine initial direction
         if(rates[bar].close > shortStop)
            prevDir = 1;
         else if(rates[bar].close < longStop)
            prevDir = -1;
         else
            prevDir = 1;  // Default bullish
         firstCalc = false;
        }

      prevLongStop  = longStop;
      prevShortStop = shortStop;
     }

   // Store results
   g_prevCeDir    = prevDir;

   // Now compute bar[1] final values
   {
      double hh = -DBL_MAX;
      double ll = DBL_MAX;
      for(int k = 1; k < 1 + period && k < ratesSize; k++)
        {
         if(rates[k].high > hh) hh = rates[k].high;
         if(rates[k].low < ll)  ll = rates[k].low;
        }

      if(hh > -DBL_MAX && ll < DBL_MAX && atrBuf[1] > 0)
        {
         g_ceLongStop  = hh - InpCEMultiplier * atrBuf[1];
         g_ceShortStop = ll + InpCEMultiplier * atrBuf[1];

         // Ratchet with previous
         if(rates[2].close > prevLongStop)
            g_ceLongStop = MathMax(g_ceLongStop, prevLongStop);
         if(rates[2].close < prevShortStop)
            g_ceShortStop = MathMin(g_ceShortStop, prevShortStop);

         // Direction
         if(rates[1].close > prevShortStop)
            g_ceDir = 1;
         else if(rates[1].close < prevLongStop)
            g_ceDir = -1;
         else
            g_ceDir = prevDir;
        }
      else
         return false;
   }

   return true;
  }

//+------------------------------------------------------------------+
//| DETECT VOLATILITY REGIME                                          |
//+------------------------------------------------------------------+
ENUM_VOLATILITY_REGIME DetectVolatilityRegime()
  {
   if(!InpUseVolatilityAdapt)
      return VOL_NORMAL;

   double atrCurrent[];
   double atrLong[];

   if(!SafeCopyBuffer(g_hATR, 0, 1, 1, atrCurrent))
      return VOL_NORMAL;
   if(!SafeCopyBuffer(g_hATR_Vol, 0, 1, 1, atrLong))
      return VOL_NORMAL;

   if(ArraySize(atrCurrent) < 1 || ArraySize(atrLong) < 1)
      return VOL_NORMAL;
   if(atrLong[0] <= 0)
      return VOL_NORMAL;

   g_atrCurrent = atrCurrent[0];
   g_atrAverage = atrLong[0];

   double ratio = g_atrCurrent / g_atrAverage;

   if(ratio >= InpVolExtremeMult)
      return VOL_EXTREME;
   if(ratio >= InpVolHighMult)
      return VOL_HIGH;
   if(ratio <= 0.6)
      return VOL_LOW;

   return VOL_NORMAL;
  }

//+------------------------------------------------------------------+
//| DETECT TREND STATE                                                |
//+------------------------------------------------------------------+
ENUM_TREND_STATE DetectTrendState()
  {
   if(!InpUseADXFilter)
      return TREND_RANGING;

   double adxMain[];
   double adxPlus[];
   double adxMinus[];

   if(!SafeCopyBuffer(g_hADX, 0, 1, 3, adxMain))
      return TREND_RANGING;
   if(!SafeCopyBuffer(g_hADX, 1, 1, 3, adxPlus))
      return TREND_RANGING;
   if(!SafeCopyBuffer(g_hADX, 2, 1, 3, adxMinus))
      return TREND_RANGING;

   if(ArraySize(adxMain) < 2 || ArraySize(adxPlus) < 2 || ArraySize(adxMinus) < 2)
      return TREND_RANGING;

   g_adxValue   = adxMain[0];
   g_adxPlusDI  = adxPlus[0];
   g_adxMinusDI = adxMinus[0];

   if(g_adxValue < InpADXThreshold)
      return TREND_RANGING;

   bool bullish = (g_adxPlusDI > g_adxMinusDI);

   if(g_adxValue >= InpADXStrongTrend)
      return bullish ? TREND_STRONG_UP : TREND_STRONG_DOWN;

   return bullish ? TREND_WEAK_UP : TREND_WEAK_DOWN;
  }

//+------------------------------------------------------------------+
//| CHECK VOLUME CONFIRMATION                                         |
//+------------------------------------------------------------------+
double GetVolumeRatio()
  {
   if(!InpUseVolumeConfirm)
      return 1.0;  // Neutral value if disabled (won't trigger bonus or penalty)

   MqlRates rates[];
   int need = InpVolumeLookback + 5;
   if(!SafeCopyRates(rates, 0, need))
      return 0;

   int rSize = ArraySize(rates);
   if(rSize < InpVolumeLookback + 2)
      return 0;

   // Calculate average volume
   double sumVol = 0;
   for(int i = 2; i < InpVolumeLookback + 2 && i < rSize; i++)
      sumVol += (double)rates[i].tick_volume;

   double avgVol = sumVol / (double)InpVolumeLookback;
   if(avgVol <= 0)
      return 0;

   g_volumeRatio = (double)rates[1].tick_volume / avgVol;
   return g_volumeRatio;
  }

//+------------------------------------------------------------------+
//| GET MOMENTUM VALUE                                                |
//+------------------------------------------------------------------+
double GetMomentum()
  {
   if(!InpUseMomentum)
      return 0;

   MqlRates rates[];
   int need = InpMomentumPeriod + 5;
   if(!SafeCopyRates(rates, 0, need))
      return 0;

   int rSize = ArraySize(rates);
   if(rSize < InpMomentumPeriod + 2)
      return 0;

   // Rate of Change
   if(rates[1 + InpMomentumPeriod].close <= 0)
      return 0;

   g_momentumValue = ((rates[1].close - rates[1 + InpMomentumPeriod].close) /
                       rates[1 + InpMomentumPeriod].close) * 100.0;
   return g_momentumValue;
  }

//+------------------------------------------------------------------+
//| DETECT RSI DIVERGENCE                                             |
//| Bullish: price makes lower low but RSI makes higher low           |
//| Bearish: price makes higher high but RSI makes lower high         |
//+------------------------------------------------------------------+
int DetectRSIDivergence()
  {
   if(!InpUseRSIDivergence)
      return 0;

   double rsi[];
   if(!SafeCopyBuffer(g_hRSI, 0, 1, InpRSIDivLookback + 2, rsi))
      return 0;

   int rsiSize = ArraySize(rsi);
   if(rsiSize < InpRSIDivLookback)
      return 0;

   MqlRates rates[];
   if(!SafeCopyRates(rates, 0, InpRSIDivLookback + 5))
      return 0;

   int rSize = ArraySize(rates);
   if(rSize < InpRSIDivLookback)
      return 0;

   // Find two most recent swing lows/highs in price and compare RSI
   int lookLeft  = 2;
   int lookRight = 2;

   int swLowIdx[];
   int swHighIdx[];
   int swLowCount  = 0;
   int swHighCount = 0;
   ArrayResize(swLowIdx, 0);
   ArrayResize(swHighIdx, 0);

   int limit = MathMin(rSize - lookLeft - 1, InpRSIDivLookback);
   for(int i = lookRight + 1; i < limit; i++)
     {
      bool isLow  = true;
      bool isHigh = true;

      for(int j = 1; j <= lookLeft; j++)
        {
         if(i + j >= rSize) { isLow = false; isHigh = false; break; }
         if(rates[i].low >= rates[i + j].low)   isLow = false;
         if(rates[i].high <= rates[i + j].high) isHigh = false;
        }
      for(int j = 1; j <= lookRight; j++)
        {
         if(i - j < 0) { isLow = false; isHigh = false; break; }
         if(rates[i].low >= rates[i - j].low)   isLow = false;
         if(rates[i].high <= rates[i - j].high) isHigh = false;
        }

      if(isLow && swLowCount < 3)
        {
         swLowCount++;
         ArrayResize(swLowIdx, swLowCount);
         swLowIdx[swLowCount - 1] = i;
        }
      if(isHigh && swHighCount < 3)
        {
         swHighCount++;
         ArrayResize(swHighIdx, swHighCount);
         swHighIdx[swHighCount - 1] = i;
        }

      if(swLowCount >= 2 && swHighCount >= 2)
         break;
     }

   // Bullish divergence: price lower low + RSI higher low
   if(swLowCount >= 2)
     {
      int idx0 = swLowIdx[0];  // More recent
      int idx1 = swLowIdx[1];  // Older
      if(idx0 < rsiSize && idx1 < rsiSize)
        {
         if(rates[idx0].low < rates[idx1].low && rsi[idx0] > rsi[idx1])
           {
            g_rsiDivSignal = 1;
            return 1;
           }
        }
     }

   // Bearish divergence: price higher high + RSI lower high
   if(swHighCount >= 2)
     {
      int idx0 = swHighIdx[0]; // More recent
      int idx1 = swHighIdx[1]; // Older
      if(idx0 < rsiSize && idx1 < rsiSize)
        {
         if(rates[idx0].high > rates[idx1].high && rsi[idx0] < rsi[idx1])
           {
            g_rsiDivSignal = -1;
            return -1;
           }
        }
     }

   g_rsiDivSignal = 0;
   return 0;
  }

//+------------------------------------------------------------------+
//| COUNT CONSECUTIVE HA CANDLES                                       |
//| Count how many recent HA bars are in the same direction            |
//+------------------------------------------------------------------+
void CountConsecutiveCandles()
  {
   g_consecBullCount = 0;
   g_consecBearCount = 0;

   if(!InpUseConsecCandles)
      return;

   int haSize = ArraySize(g_haClose);
   if(haSize < 8)
      return;

   // Count consecutive bullish candles from bar[1] backward
   for(int i = 1; i < haSize && i < 10; i++)
     {
      if(g_haClose[i] > g_haOpen[i])
         g_consecBullCount++;
      else
         break;
     }

   // Count consecutive bearish candles from bar[1] backward
   for(int i = 1; i < haSize && i < 10; i++)
     {
      if(g_haClose[i] < g_haOpen[i])
         g_consecBearCount++;
      else
         break;
     }
  }

//+------------------------------------------------------------------+
//| DETECT MARKET STRUCTURE (Higher Highs / Lower Lows)               |
//+------------------------------------------------------------------+
int DetectMarketStructure()
  {
   if(!InpUseStructure)
      return 0;

   MqlRates rates[];
   int need = InpStructureLookback + 5;
   if(!SafeCopyRates(rates, 0, need))
      return 0;

   int rSize = ArraySize(rates);
   if(rSize < 20)
      return 0;

   // Find swing points (simple pivot detection)
   double swHighs[];
   double swLows[];
   int    swHighCount = 0;
   int    swLowCount  = 0;

   ArrayResize(swHighs, 0);
   ArrayResize(swLows, 0);

   int lookLeft  = 3;
   int lookRight = 3;

   for(int i = lookRight + 1; i < rSize - lookLeft - 1 && i < InpStructureLookback; i++)
     {
      bool isSwingHigh = true;
      bool isSwingLow  = true;

      for(int j = 1; j <= lookLeft; j++)
        {
         if(i + j >= rSize) { isSwingHigh = false; isSwingLow = false; break; }
         if(rates[i].high <= rates[i + j].high) isSwingHigh = false;
         if(rates[i].low >= rates[i + j].low)   isSwingLow = false;
        }
      for(int j = 1; j <= lookRight; j++)
        {
         if(i - j < 0) { isSwingHigh = false; isSwingLow = false; break; }
         if(rates[i].high <= rates[i - j].high) isSwingHigh = false;
         if(rates[i].low >= rates[i - j].low)   isSwingLow = false;
        }

      if(isSwingHigh)
        {
         swHighCount++;
         ArrayResize(swHighs, swHighCount);
         swHighs[swHighCount - 1] = rates[i].high;
        }
      if(isSwingLow)
        {
         swLowCount++;
         ArrayResize(swLows, swLowCount);
         swLows[swLowCount - 1] = rates[i].low;
        }
     }

   // Analyze structure: need at least 2 swing points each
   if(swHighCount < 2 || swLowCount < 2)
      return 0;

   // swHighs[0] is most recent swing high, swHighs[1] is previous
   bool higherHighs = (swHighs[0] > swHighs[1]);
   bool higherLows  = (swLows[0] > swLows[1]);
   bool lowerHighs  = (swHighs[0] < swHighs[1]);
   bool lowerLows   = (swLows[0] < swLows[1]);

   // Enhanced: if 3 swing points available, require 2 consecutive confirmations
   if(swHighCount >= 3 && swLowCount >= 3)
     {
      bool hh2 = (swHighs[1] > swHighs[2]);  // Previous pair also HH
      bool hl2 = (swLows[1] > swLows[2]);     // Previous pair also HL
      bool lh2 = (swHighs[1] < swHighs[2]);
      bool ll2 = (swLows[1] < swLows[2]);

      // Strong structure: 2 consecutive HH/HL or LH/LL
      if(higherHighs && higherLows && hh2 && hl2)
        {
         g_structureDir = 1;
         return 2;  // Strong bullish structure (return 2 for extra weight)
        }
      if(lowerHighs && lowerLows && lh2 && ll2)
        {
         g_structureDir = -1;
         return -2; // Strong bearish structure
        }
     }

   if(higherHighs && higherLows)
     {
      g_structureDir = 1;   // Bullish structure
      return 1;
     }
   if(lowerHighs && lowerLows)
     {
      g_structureDir = -1;  // Bearish structure
      return -1;
     }

   g_structureDir = 0;
   return 0;
  }

//+------------------------------------------------------------------+
//| HIGHER TIMEFRAME TREND CHECK                                      |
//+------------------------------------------------------------------+
int GetHTFTrend()
  {
   if(!InpUseMultiTF)
      return 0;  // Neutral = no filter

   double htfRSI[];
   if(!SafeCopyBuffer(g_hHTF_RSI, 0, 1, 3, htfRSI))
      return 0;

   if(ArraySize(htfRSI) < 2)
      return 0;

   int rsiVote = 0;
   if(htfRSI[0] > 55)
      rsiVote = 1;
   else if(htfRSI[0] < 45)
      rsiVote = -1;

   // Also check HTF RSI momentum (is RSI increasing or decreasing?)
   int rsiMomVote = 0;
   if(htfRSI[0] > htfRSI[1])
      rsiMomVote = 1;
   else if(htfRSI[0] < htfRSI[1])
      rsiMomVote = -1;

   // HTF ATR direction: check if HTF volatility is expanding (trending)
   double htfATR[];
   if(SafeCopyBuffer(g_hHTF_ATR, 0, 1, 3, htfATR) && ArraySize(htfATR) >= 2)
     {
      // If HTF ATR is expanding, the trend signal is more reliable
      if(htfATR[0] > htfATR[1] && rsiVote != 0)
         return rsiVote;  // Expanding ATR confirms trend direction
     }

   // If RSI level and RSI momentum agree, that's a strong signal
   if(rsiVote != 0 && rsiMomVote == rsiVote)
      return rsiVote;

   // RSI level alone (weaker)
   if(htfRSI[0] > 58)
      return 1;
   if(htfRSI[0] < 42)
      return -1;

   return 0;
  }

//+------------------------------------------------------------------+
//| CONFLUENCE-BASED SIGNAL GENERATION                                |
//+------------------------------------------------------------------+
int GetSignal()
  {
   int minBars = InpCEPeriod;
   if(InpZLSMAPeriod > minBars) minBars = InpZLSMAPeriod;
   if(InpStructureLookback > minBars) minBars = InpStructureLookback;
   int needBars = minBars * 3 + 50;

   MqlRates rates[];
   if(!SafeCopyRates(rates, 0, needBars))
      return 0;

   int ratesSize = ArraySize(rates);
   if(ratesSize < 10)
      return 0;

   //--- Build Heikin Ashi cache
   if(InpUseHeikinAshi)
     {
      if(!BuildHeikinAshiCache(needBars))
         return 0;
     }

   //--- RSI
   double rsi[];
   if(!SafeCopyBuffer(g_hRSI, 0, 0, 5, rsi))
      return 0;
   if(ArraySize(rsi) < 4)
      return 0;

   //--- Bollinger Bands
   double bbUpper[], bbMiddle[], bbLower[];
   if(!SafeCopyBuffer(g_hBB, 1, 0, 5, bbUpper))   return 0;
   if(!SafeCopyBuffer(g_hBB, 0, 0, 5, bbMiddle))  return 0;
   if(!SafeCopyBuffer(g_hBB, 2, 0, 5, bbLower))   return 0;
   if(ArraySize(bbUpper) < 3 || ArraySize(bbLower) < 3)
      return 0;

   //--- Chandelier Exit
   if(!CalcChandelierExit())
      return 0;

   //--- ZLSMA (3 bars for slope analysis)
   g_zlsma1 = CalcZLSMA(1);
   g_zlsma2 = CalcZLSMA(2);
   g_zlsma3 = CalcZLSMA(3);
   if(g_zlsma1 == 0 || g_zlsma2 == 0 || g_zlsma3 == 0)
      return 0;

   //--- Volatility regime
   g_volRegime = DetectVolatilityRegime();

   //--- Trend state
   g_trendState = DetectTrendState();

   //--- Market structure
   int structure = DetectMarketStructure();

   //--- Volume
   double volRatio = GetVolumeRatio();

   //--- Momentum
   double momentum = GetMomentum();

   //--- HTF trend
   int htfTrend = GetHTFTrend();

   //--- RSI Divergence
   int rsiDivergence = DetectRSIDivergence();

   //--- Consecutive candles (uses HA cache if available)
   if(InpUseHeikinAshi)
      CountConsecutiveCandles();

   //--- Price data
   double close1 = 0, close2 = 0, close3 = 0;
   double open1 = 0, open2 = 0;
   double high1 = 0, low1 = 0;

   if(InpUseHeikinAshi && ArraySize(g_haClose) >= 4)
     {
      close1 = g_haClose[1]; close2 = g_haClose[2]; close3 = g_haClose[3];
      open1  = g_haOpen[1];  open2  = g_haOpen[2];
      high1  = g_haHigh[1];  low1   = g_haLow[1];
     }
   else
     {
      close1 = rates[1].close; close2 = rates[2].close; close3 = rates[3].close;
      open1  = rates[1].open;  open2  = rates[2].open;
      high1  = rates[1].high;  low1   = rates[1].low;
     }

   if(close1 == 0 || close2 == 0 || open1 == 0)
      return 0;

   //=================================================================
   // CONFLUENCE SCORING SYSTEM
   // Each factor adds to the score. Minimum score required to trade.
   //=================================================================
   int buyScore  = 0;
   int sellScore = 0;
   g_signalDetails = "";

   //--- Factor 1: Chandelier Exit Direction + Transition (weight: 1-2)
   bool ceTurnedBull = (g_ceDir == 1 && g_prevCeDir == -1);
   bool ceTurnedBear = (g_ceDir == -1 && g_prevCeDir == 1);

   if(ceTurnedBull)
     {
      buyScore += 2;
      g_signalDetails += "CE_FLIP+ ";
     }
   else if(g_ceDir == 1)
     {
      buyScore += 1;
      g_signalDetails += "CE+ ";
     }

   if(ceTurnedBear)
     {
      sellScore += 2;
      g_signalDetails += "CE_FLIP- ";
     }
   else if(g_ceDir == -1)
     {
      sellScore += 1;
      g_signalDetails += "CE- ";
     }

   //--- Factor 2: ZLSMA Trend + Slope (weight: 1-2)
   double zlsmaSlope1 = g_zlsma1 - g_zlsma2;
   double zlsmaSlope2 = g_zlsma2 - g_zlsma3;
   bool   zlsmaAccelUp   = (zlsmaSlope1 > 0 && zlsmaSlope1 > zlsmaSlope2);
   bool   zlsmaAccelDown = (zlsmaSlope1 < 0 && zlsmaSlope1 < zlsmaSlope2);

   if(close1 > g_zlsma1 && g_zlsma1 > g_zlsma2)
     {
      buyScore += 1;
      g_signalDetails += "ZL+ ";
      if(zlsmaAccelUp)
        {
         buyScore += 1;
         g_signalDetails += "ZLaccel+ ";
        }
     }

   if(close1 < g_zlsma1 && g_zlsma1 < g_zlsma2)
     {
      sellScore += 1;
      g_signalDetails += "ZL- ";
      if(zlsmaAccelDown)
        {
         sellScore += 1;
         g_signalDetails += "ZLaccel- ";
        }
     }

   //--- Factor 3: Candle Pattern (weight: 0-1)
   bool bullCandle = (close1 > open1);
   bool bearCandle = (close1 < open1);

   // Enhanced: check candle body strength
   double bodySize = MathAbs(close1 - open1);
   double wickSize = high1 - low1;
   bool   strongBullCandle = bullCandle && (wickSize > 0) && (bodySize / wickSize > 0.5);
   bool   strongBearCandle = bearCandle && (wickSize > 0) && (bodySize / wickSize > 0.5);

   if(strongBullCandle)
     {
      buyScore += 1;
      g_signalDetails += "CANDLE+ ";
     }
   // Weak candle: no score added (direction noted but not reliable enough)

   if(strongBearCandle)
     {
      sellScore += 1;
      g_signalDetails += "CANDLE- ";
     }
   // Weak candle: no score added

   //--- Factor 4: RSI (weight: 1)
   if(InpUseRSIFilter)
     {
      // RSI confirms direction without being overbought/oversold against us
      if(rsi[1] > 50 && rsi[1] < InpRSIOverbought)
        {
         buyScore += 1;
         g_signalDetails += "RSI+ ";
        }
      if(rsi[1] < 50 && rsi[1] > InpRSIOversold)
        {
         sellScore += 1;
         g_signalDetails += "RSI- ";
        }

      // RSI divergence / reversal from extremes (bonus signal)
      if(rsi[2] < InpRSIOversold && rsi[1] > InpRSIOversold)
        {
         buyScore += 1;
         g_signalDetails += "RSI_REV+ ";
        }
      if(rsi[2] > InpRSIOverbought && rsi[1] < InpRSIOverbought)
        {
         sellScore += 1;
         g_signalDetails += "RSI_REV- ";
        }

      // PENALTY: Trading into extreme RSI
      if(rsi[1] >= InpRSIOverbought)
        {
         buyScore -= 2;
         g_signalDetails += "RSI_OB! ";
        }
      if(rsi[1] <= InpRSIOversold)
        {
         sellScore -= 2;
         g_signalDetails += "RSI_OS! ";
        }
     }

   //--- Factor 5: Bollinger Bands (weight: 1)
   if(InpUseBBFilter)
     {
      // Price near lower band = buy opportunity
      if(close1 <= bbLower[1] || (close2 < bbLower[2] && close1 > bbLower[1]))
        {
         buyScore += 1;
         g_signalDetails += "BB+ ";
        }
      // Price near upper band = sell opportunity
      if(close1 >= bbUpper[1] || (close2 > bbUpper[2] && close1 < bbUpper[1]))
        {
         sellScore += 1;
         g_signalDetails += "BB- ";
        }

      // PENALTY: Breakout against position
      if(close1 > bbUpper[1] && close2 > bbUpper[2])
         sellScore -= 1;
      if(close1 < bbLower[1] && close2 < bbLower[2])
         buyScore -= 1;
     }

   //--- Factor 6: ADX / Trend Strength (weight: 1)
   if(InpUseADXFilter)
     {
      if(g_trendState == TREND_STRONG_UP || g_trendState == TREND_WEAK_UP)
        {
         buyScore += 1;
         g_signalDetails += "ADX+ ";
        }
      if(g_trendState == TREND_STRONG_DOWN || g_trendState == TREND_WEAK_DOWN)
        {
         sellScore += 1;
         g_signalDetails += "ADX- ";
        }

      // PENALTY: Trading against strong trend
      if(g_trendState == TREND_STRONG_UP)
         sellScore -= 1;
      if(g_trendState == TREND_STRONG_DOWN)
         buyScore -= 1;

      // PENALTY: No trend (ranging) - penalize only the weaker side
      if(g_trendState == TREND_RANGING)
        {
         // In ranging markets, only reduce the side with less conviction
         // instead of penalizing both sides which makes confluence unreachable
         if(buyScore >= sellScore && sellScore > 0)
            sellScore -= 1;
         else if(sellScore > buyScore && buyScore > 0)
            buyScore -= 1;
         g_signalDetails += "RANGE! ";
        }
     }

   //--- Factor 7: Volume Confirmation (weight: 1)
   if(InpUseVolumeConfirm)
     {
      if(volRatio >= InpVolumeMinMult)
        {
         if(buyScore > sellScore)
           {
            buyScore += 1;
            g_signalDetails += "VOL+ ";
           }
         else if(sellScore > buyScore)
           {
            sellScore += 1;
            g_signalDetails += "VOL- ";
           }
        }
      else if(volRatio < 0.7)
        {
         // Low volume - reduce confidence on weaker side only
         // Avoids double-penalty with VOL_LOW regime adjustment
         if(buyScore >= sellScore && sellScore > 0)
            sellScore -= 1;
         else if(sellScore > buyScore && buyScore > 0)
            buyScore -= 1;
         g_signalDetails += "LOWVOL! ";
        }
     }

   //--- Factor 8: Momentum (weight: 1) - Adaptive thresholds
   if(InpUseMomentum)
     {
      // Scale momentum threshold with volatility
      double momThreshold = 0.1;
      if(g_volRegime == VOL_HIGH)
         momThreshold = 0.15;
      else if(g_volRegime == VOL_EXTREME)
         momThreshold = 0.25;
      else if(g_volRegime == VOL_LOW)
         momThreshold = 0.05;

      if(momentum > momThreshold)
        {
         buyScore += 1;
         g_signalDetails += "MOM+ ";
        }
      if(momentum < -momThreshold)
        {
         sellScore += 1;
         g_signalDetails += "MOM- ";
        }

      // PENALTY: Momentum fading (was strong, now weakening)
      if(momentum > 0 && momentum < momThreshold * 0.5 && g_trendState == TREND_WEAK_UP)
        {
         buyScore -= 1;
         g_signalDetails += "MOMFADE! ";
        }
      if(momentum < 0 && momentum > -momThreshold * 0.5 && g_trendState == TREND_WEAK_DOWN)
        {
         sellScore -= 1;
         g_signalDetails += "MOMFADE! ";
        }
     }

   //--- Factor 9: Market Structure (weight: 1-2, enhanced with 3-swing)
   if(InpUseStructure)
     {
      // structure returns 2/-2 for strong (3-swing confirmed), 1/-1 for normal
      if(structure >= 2)
        {
         buyScore += 2;
         g_signalDetails += "STRUCT++ ";
        }
      else if(structure == 1)
        {
         buyScore += 1;
         g_signalDetails += "STRUCT+ ";
        }
      if(structure <= -2)
        {
         sellScore += 2;
         g_signalDetails += "STRUCT-- ";
        }
      else if(structure == -1)
        {
         sellScore += 1;
         g_signalDetails += "STRUCT- ";
        }
     }

   //--- Factor 10: Higher Timeframe (weight: 1)
   if(InpUseMultiTF)
     {
      if(htfTrend == 1)
        {
         buyScore += 1;
         sellScore -= 1;
         g_signalDetails += "HTF+ ";
        }
      if(htfTrend == -1)
        {
         sellScore += 1;
         buyScore -= 1;
         g_signalDetails += "HTF- ";
        }
     }

   //--- Factor 11: RSI Divergence (weight: 1-2)
   if(InpUseRSIDivergence)
     {
      if(rsiDivergence == 1)
        {
         buyScore += 2;
         g_signalDetails += "RSI_DIV+ ";
         // Divergence against strong sell pressure is very valuable
         if(sellScore > buyScore)
            buyScore += 1;
        }
      if(rsiDivergence == -1)
        {
         sellScore += 2;
         g_signalDetails += "RSI_DIV- ";
         if(buyScore > sellScore)
            sellScore += 1;
        }
     }

   //--- Factor 12: Consecutive HA Candles (weight: 1)
   if(InpUseConsecCandles && InpUseHeikinAshi)
     {
      if(g_consecBullCount >= InpConsecCandleMin)
        {
         buyScore += 1;
         g_signalDetails += StringFormat("HA%d+ ", g_consecBullCount);
        }
      if(g_consecBearCount >= InpConsecCandleMin)
        {
         sellScore += 1;
         g_signalDetails += StringFormat("HA%d- ", g_consecBearCount);
        }

      // PENALTY: Exhaustion after too many consecutive candles (>= 6)
      if(g_consecBullCount >= 6)
        {
         buyScore -= 1;
         g_signalDetails += "HA_EXHAUST! ";
        }
      if(g_consecBearCount >= 6)
        {
         sellScore -= 1;
         g_signalDetails += "HA_EXHAUST! ";
        }
     }

   //--- Factor 13: ZLSMA Crossover (weight: 1)
   // Price crossing ZLSMA is a momentum shift signal
   {
      bool zlsmaCrossUp   = (close2 < g_zlsma2 && close1 > g_zlsma1);
      bool zlsmaCrossDown = (close2 > g_zlsma2 && close1 < g_zlsma1);

      if(zlsmaCrossUp)
        {
         buyScore += 1;
         g_signalDetails += "ZLx+ ";
        }
      if(zlsmaCrossDown)
        {
         sellScore += 1;
         g_signalDetails += "ZLx- ";
        }
   }

   //--- Volatility regime adjustments
   if(g_volRegime == VOL_EXTREME)
     {
      // Reduce all signals in extreme volatility
      buyScore  -= 1;
      sellScore -= 1;
      g_signalDetails += "EXTVOL! ";
     }
   else if(g_volRegime == VOL_LOW)
     {
      // Low volatility - penalize weaker side only (less harsh than extreme)
      if(buyScore >= sellScore && sellScore > 0)
         sellScore -= 1;
      else if(sellScore > buyScore && buyScore > 0)
         buyScore -= 1;
      g_signalDetails += "LOWVOL_R! ";
     }

   //--- Floor at zero
   if(buyScore < 0)  buyScore = 0;
   if(sellScore < 0) sellScore = 0;

   //--- Store confluence score
   g_confluenceScore = MathMax(buyScore, sellScore);
   g_lastConfluence  = g_confluenceScore;

   //--- Determine signal
   int signal = 0;

   if(buyScore >= InpMinConfluence && buyScore > sellScore)
      signal = 1;
   else if(sellScore >= InpMinConfluence && sellScore > buyScore)
      signal = -1;

   // Conflicting signals cancel out
   if(buyScore >= InpMinConfluence && sellScore >= InpMinConfluence)
     {
      if(MathAbs(buyScore - sellScore) < 2)
        {
         PrintFormat("[INFO] Conflicting signals cancelled: Buy=%d Sell=%d (diff < 2) | %s",
                     buyScore, sellScore, g_signalDetails);
         signal = 0;  // Too close - no trade
        }
     }

   // Log when score exists but below minimum (helps diagnose no-trade issues)
   if(signal == 0 && (buyScore > 0 || sellScore > 0))
      PrintFormat("[INFO] Score below minimum: Buy=%d Sell=%d Min=%d | %s",
                  buyScore, sellScore, InpMinConfluence, g_signalDetails);

   //--- Reverse if enabled
   if(InpReverseSignal && signal != 0)
      signal = -signal;

   return signal;
  }

//+------------------------------------------------------------------+
//| CALCULATE LOT SIZE                                                |
//+------------------------------------------------------------------+
double CalcLotSize(double slPoints)
  {
   symInfo.Name(_Symbol);
   symInfo.RefreshRates();

   double tickValue = symInfo.TickValue();
   double tickSize  = symInfo.TickSize();
   double point     = symInfo.Point();
   double minLot    = symInfo.LotsMin();
   double maxLot    = symInfo.LotsMax();
   double lotStep   = symInfo.LotsStep();
   double lotSize   = minLot;

   if(tickValue <= 0 || tickSize <= 0 || point <= 0 || lotStep <= 0)
      return minLot;

   switch(InpRiskMode)
     {
      case RISK_FIXED_LOT:
         lotSize = InpRiskValue;
         break;

      case RISK_PERCENT_BALANCE:
        {
         double riskAmount = accInfo.Balance() * InpRiskValue / 100.0;
         if(slPoints > 0)
            lotSize = riskAmount / ((slPoints * point / tickSize) * tickValue);
         else
            lotSize = minLot;
        }
      break;

      case RISK_PERCENT_EQUITY:
        {
         double riskAmount = accInfo.Equity() * InpRiskValue / 100.0;
         if(slPoints > 0)
            lotSize = riskAmount / ((slPoints * point / tickSize) * tickValue);
         else
            lotSize = minLot;
        }
      break;
     }

   // Volatility-based lot reduction
   if(InpUseVolatilityAdapt)
     {
      if(g_volRegime == VOL_HIGH)
         lotSize *= 0.75;
      else if(g_volRegime == VOL_EXTREME)
         lotSize *= 0.5;
     }

   if(lotSize < minLot) lotSize = minLot;
   if(lotSize > maxLot) lotSize = maxLot;
   lotSize = MathFloor(lotSize / lotStep) * lotStep;

   // Determine decimal precision from lotStep (e.g. 0.01→2, 0.001→3)
   int lotPrecision = 2;
   if(lotStep >= 1.0)        lotPrecision = 0;
   else if(lotStep >= 0.1)   lotPrecision = 1;
   else if(lotStep >= 0.01)  lotPrecision = 2;
   else if(lotStep >= 0.001) lotPrecision = 3;
   else                      lotPrecision = 4;

   lotSize = NormalizeDouble(lotSize, lotPrecision);
   if(lotSize < minLot) lotSize = minLot;

   return lotSize;
  }

//+------------------------------------------------------------------+
//| CALCULATE STOP LOSS                                               |
//+------------------------------------------------------------------+
double CalcStopLoss(bool isBuy)
  {
   symInfo.Name(_Symbol);
   symInfo.RefreshRates();

   double point  = symInfo.Point();
   int    digits = (int)symInfo.Digits();
   double price  = isBuy ? symInfo.Ask() : symInfo.Bid();
   double sl     = 0;

   if(point <= 0 || price <= 0)
      return 0;

   switch(InpSLType)
     {
      case SL_FIXED_POINTS:
         sl = isBuy ? price - InpSLFixedPoints * point : price + InpSLFixedPoints * point;
         break;

      case SL_ATR_BASED:
        {
         double atrBuf[];
         if(SafeCopyBuffer(g_hATR, 0, 1, 1, atrBuf) && ArraySize(atrBuf) >= 1 && atrBuf[0] > 0)
           {
            double atrSL = atrBuf[0] * InpATRMultSL;

            // Adapt ATR multiplier to volatility regime
            if(InpUseVolatilityAdapt)
              {
               if(g_volRegime == VOL_HIGH)
                  atrSL *= 1.3;
               else if(g_volRegime == VOL_EXTREME)
                  atrSL *= 1.6;
               else if(g_volRegime == VOL_LOW)
                  atrSL *= 0.8;
              }

            sl = isBuy ? price - atrSL - InpSLDeviation * point
                 : price + atrSL + InpSLDeviation * point;
           }
         else
            sl = isBuy ? price - InpSLFixedPoints * point : price + InpSLFixedPoints * point;
        }
      break;

      case SL_SWING_HIGH_LOW:
        {
         int need = InpSLLookback + 2;
         MqlRates swRates[];
         if(SafeCopyRates(swRates, 1, need) && ArraySize(swRates) >= InpSLLookback)
           {
            int swSize = ArraySize(swRates);
            if(isBuy)
              {
               double lowestLow = DBL_MAX;
               for(int i = 0; i < InpSLLookback && i < swSize; i++)
                  if(swRates[i].low < lowestLow) lowestLow = swRates[i].low;
               if(lowestLow < DBL_MAX)
                  sl = lowestLow - InpSLDeviation * point;
              }
            else
              {
               double highestHigh = -DBL_MAX;
               for(int i = 0; i < InpSLLookback && i < swSize; i++)
                  if(swRates[i].high > highestHigh) highestHigh = swRates[i].high;
               if(highestHigh > -DBL_MAX)
                  sl = highestHigh + InpSLDeviation * point;
              }
           }

         if(sl <= 0)
            sl = isBuy ? price - InpSLFixedPoints * point : price + InpSLFixedPoints * point;
        }
      break;

      case SL_CHANDELIER:
        {
         if(g_ceLongStop > 0 && g_ceShortStop > 0)
            sl = isBuy ? g_ceLongStop - InpSLDeviation * point
                 : g_ceShortStop + InpSLDeviation * point;
         else
            sl = isBuy ? price - InpSLFixedPoints * point : price + InpSLFixedPoints * point;
        }
      break;
     }

   if(sl <= 0)
      sl = isBuy ? price - InpSLFixedPoints * point : price + InpSLFixedPoints * point;

   return NormalizeDouble(sl, digits);
  }

//+------------------------------------------------------------------+
//| CALCULATE ADAPTIVE TAKE PROFIT                                    |
//+------------------------------------------------------------------+
double CalcTakeProfit(bool isBuy, double price, double sl)
  {
   if(InpTPCoefficient <= 0 || sl <= 0 || price <= 0)
      return 0;

   symInfo.Name(_Symbol);
   int digits = (int)symInfo.Digits();

   double slDist = MathAbs(price - sl);
   double tpCoeff = InpTPCoefficient;

   // Adaptive TP based on conditions
   if(InpAdaptiveTP)
     {
      // Strong trend = wider TP
      if(g_trendState == TREND_STRONG_UP || g_trendState == TREND_STRONG_DOWN)
         tpCoeff *= 1.5;
      // Weak trend = tighter TP
      else if(g_trendState == TREND_WEAK_UP || g_trendState == TREND_WEAK_DOWN)
         tpCoeff *= 1.0;
      // Ranging = tighter TP (take what you can)
      else if(g_trendState == TREND_RANGING)
         tpCoeff *= 0.75;

      // High confluence = wider TP (more confident)
      if(g_confluenceScore >= 5)
         tpCoeff *= 1.2;

      // Volatility adjustment
      if(g_volRegime == VOL_HIGH)
         tpCoeff *= 1.2;
      else if(g_volRegime == VOL_LOW)
         tpCoeff *= 0.8;
     }

   double tp = 0;
   if(isBuy)
      tp = price + tpCoeff * slDist;
   else
      tp = price - tpCoeff * slDist;

   return NormalizeDouble(tp, digits);
  }

//| COUNT POSITIONS                                                   |
//+------------------------------------------------------------------+
int CountPositions(int direction = 0)
  {
   int count = 0;
   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; i--)
     {
      if(posInfo.SelectByIndex(i))
        {
         if(posInfo.Magic() == InpMagicNumber && posInfo.Symbol() == _Symbol)
           {
            if(direction == 0)
               count++;
            else if(direction == 1 && posInfo.PositionType() == POSITION_TYPE_BUY)
               count++;
            else if(direction == -1 && posInfo.PositionType() == POSITION_TYPE_SELL)
               count++;
           }
        }
     }
   return count;
  }

//+------------------------------------------------------------------+
double GetTotalProfit()
  {
   double profit = 0;
   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; i--)
     {
      if(posInfo.SelectByIndex(i))
        {
         if(posInfo.Magic() == InpMagicNumber && posInfo.Symbol() == _Symbol)
            profit += posInfo.Profit() + posInfo.Swap() + posInfo.Commission();
        }
     }
   return profit;
  }

//+------------------------------------------------------------------+
//| GET LAST ENTRY PRICE FOR GRID                                     |
//+------------------------------------------------------------------+
double GetLastEntryPrice(ENUM_POSITION_TYPE type)
  {
   datetime latestTime = 0;
   double   lastPrice  = 0;
   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; i--)
     {
      if(posInfo.SelectByIndex(i))
        {
         if(posInfo.Magic() == InpMagicNumber && posInfo.Symbol() == _Symbol && posInfo.PositionType() == type)
           {
            if(posInfo.Time() > latestTime)
              {
               latestTime = posInfo.Time();
               lastPrice  = posInfo.PriceOpen();
              }
           }
        }
     }
   return lastPrice;
  }

//+------------------------------------------------------------------+
//| GET AVERAGE ENTRY PRICE                                           |
//+------------------------------------------------------------------+
double GetAverageEntryPrice(ENUM_POSITION_TYPE type)
  {
   double totalLots  = 0;
   double weightedPrice = 0;
   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; i--)
     {
      if(posInfo.SelectByIndex(i))
        {
         if(posInfo.Magic() == InpMagicNumber && posInfo.Symbol() == _Symbol && posInfo.PositionType() == type)
           {
            weightedPrice += posInfo.PriceOpen() * posInfo.Volume();
            totalLots     += posInfo.Volume();
           }
        }
     }
   if(totalLots <= 0)
      return 0;
   return weightedPrice / totalLots;
  }

//+------------------------------------------------------------------+
//| CALCULATE GRID LOT SIZE                                           |
//+------------------------------------------------------------------+
double CalcGridLots(int level)
  {
   symInfo.Name(_Symbol);
   symInfo.RefreshRates();

   double baseLots = CalcLotSize(InpSLFixedPoints > 0 ? (double)InpSLFixedPoints : 200.0);
   double gridLots = baseLots;
   double minLot   = symInfo.LotsMin();
   double maxLot   = symInfo.LotsMax();
   double lotStep  = symInfo.LotsStep();

   double mult = InpGridVolMultiplier;
   if(mult <= 0) mult = 1.0;

   switch(InpGridMode)
     {
      case GRID_SAME_DIRECTION:
         gridLots = baseLots * mult;
         break;
      case GRID_MARTINGALE:
        {
         double mMult = mult > 1.0 ? mult : 1.5;
         // Cap at level 6 to prevent exponential blowup (1.5^6=11.4x vs 1.5^10=57.7x)
         int    cappedLevel = MathMin(level, 6);
         gridLots = baseLots * MathPow(mMult, (double)cappedLevel);
        }
      break;
      case GRID_ANTI_MARTINGALE:
        {
         double aMult = mult > 1.0 ? mult : 1.5;
         gridLots = level > 0 ? baseLots / MathPow(aMult, (double)level) : baseLots;
        }
      break;
      default:
         gridLots = baseLots;
         break;
     }

   // Volatility-based grid lot reduction:
   // High vol: 80% at level 0, decreasing by 3% per level (floor at 50%)
   // Extreme vol: 60% at level 0, decreasing by 5% per level (floor at 30%)
   if(InpUseVolatilityAdapt)
     {
      if(g_volRegime == VOL_HIGH)
         gridLots *= MathMax(0.5, 0.8 - level * 0.03);
      else if(g_volRegime == VOL_EXTREME)
         gridLots *= MathMax(0.3, 0.6 - level * 0.05);
     }

   if(gridLots < minLot) gridLots = minLot;
   if(gridLots > maxLot) gridLots = maxLot;
   if(lotStep > 0) gridLots = MathFloor(gridLots / lotStep) * lotStep;
   if(gridLots < minLot) gridLots = minLot;

   return NormalizeDouble(gridLots, 2);
  }

//+------------------------------------------------------------------+
//| OPEN TRADE                                                        |
//+------------------------------------------------------------------+
bool OpenTrade(int signal)
  {
   if(!InpAllowNewPos)
      return false;
   if(CountPositions() >= InpMaxPositions)
      return false;

   symInfo.Name(_Symbol);
   symInfo.RefreshRates();

   if(InpSpreadLimit >= 0 && (int)symInfo.Spread() > InpSpreadLimit)
     {
      PrintFormat("[BLOCK] Spread too high: %d > %d pts — trade skipped",
                  (int)symInfo.Spread(), InpSpreadLimit);
      return false;
     }

   if(InpMarginLevelLimit > 0 && PositionsTotal() > 0)
     {
      double ml = accInfo.MarginLevel();
      if(ml > 0 && ml < InpMarginLevelLimit)
        {
         PrintFormat("[BLOCK] Margin level too low: %.1f%% < %.1f%% — trade skipped",
                     ml, InpMarginLevelLimit);
         return false;
        }
     }

   bool   isBuy  = (signal == 1);
   double price  = isBuy ? symInfo.Ask() : symInfo.Bid();
   int    digits = (int)symInfo.Digits();
   double point  = symInfo.Point();
   double sl     = CalcStopLoss(isBuy);

   if(sl <= 0 || price <= 0 || point <= 0)
      return false;

   double slDist = MathAbs(price - sl) / point;
   double tp     = CalcTakeProfit(isBuy, price, sl);
   double lots   = CalcLotSize(slDist);

   // Free margin check: avoid silent rejection on small accounts (e.g. $100)
   double marginRequired = 0;
   ENUM_ORDER_TYPE orderType = isBuy ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   if(OrderCalcMargin(orderType, _Symbol, lots, price, marginRequired))
     {
      if(accInfo.FreeMargin() < marginRequired)
        {
         PrintFormat("[WARN] Insufficient free margin: need=%.2f, free=%.2f, lots=%.3f. "
                     "Balance=%.2f - consider reducing InpRiskValue or using a larger account.",
                     marginRequired, accInfo.FreeMargin(), lots, accInfo.Balance());
         return false;
        }
     }

   price = NormalizeDouble(price, digits);
   sl    = NormalizeDouble(sl, digits);
   if(tp > 0) tp = NormalizeDouble(tp, digits);

   string comment = StringFormat("%s|C%d", InpEAComment, g_confluenceScore);

   bool result = false;
   if(isBuy)
      result = trade.Buy(lots, _Symbol, price, sl, tp, comment);
   else
      result = trade.Sell(lots, _Symbol, price, sl, tp, comment);

   if(result)
      PrintFormat("[TRADE] %s | Price: %.5f | SL: %.5f | TP: %.5f | Lots: %.3f | Confluence: %d | %s",
                  isBuy ? "BUY" : "SELL", price, sl, tp, lots, g_confluenceScore, g_signalDetails);
   else
      PrintFormat("[ERROR] Trade failed: %d - %s | Lots: %.3f | Balance: %.2f | FreeMargin: %.2f",
                  GetLastError(), trade.ResultComment(), lots, accInfo.Balance(), accInfo.FreeMargin());

   return result;
  }

//+------------------------------------------------------------------+
//| SMART GRID: Multi-factor value zone check for grid entry          |
//+------------------------------------------------------------------+
bool IsGridValueZone(bool isBuy)
  {
   if(!InpGridSmartEntry)
      return true;  // Always allow if smart entry disabled

   int valueScore = 0;  // Need >= 2 to pass

   // Factor 1: RSI check - price has pulled back
   double rsi[];
   if(SafeCopyBuffer(g_hRSI, 0, 1, 2, rsi) && ArraySize(rsi) >= 2)
     {
      if(isBuy)
        {
         if(rsi[0] < 45) valueScore += 2;       // Oversold = great value
         else if(rsi[0] < 55) valueScore += 1;   // Neutral = ok
         else if(rsi[0] > 70) return false;       // Overbought = never add buy grid
        }
      else
        {
         if(rsi[0] > 55) valueScore += 2;       // Overbought = great value
         else if(rsi[0] > 45) valueScore += 1;   // Neutral = ok
         else if(rsi[0] < 30) return false;       // Oversold = never add sell grid
        }
     }
   else
      valueScore += 1;  // No data = assume ok

   // Factor 2: Bollinger Band proximity - price near value band
   double bbUpper[], bbLower[], bbMiddle[];
   if(SafeCopyBuffer(g_hBB, 1, 1, 1, bbUpper) && ArraySize(bbUpper) >= 1 &&
      SafeCopyBuffer(g_hBB, 2, 1, 1, bbLower) && ArraySize(bbLower) >= 1 &&
      SafeCopyBuffer(g_hBB, 0, 1, 1, bbMiddle) && ArraySize(bbMiddle) >= 1)
     {
      symInfo.RefreshRates();
      double price = isBuy ? symInfo.Ask() : symInfo.Bid();

      if(isBuy)
        {
         // Buy grid value: price near or below lower BB or below middle
         if(price <= bbLower[0]) valueScore += 2;
         else if(price <= bbMiddle[0]) valueScore += 1;
        }
      else
        {
         // Sell grid value: price near or above upper BB or above middle
         if(price >= bbUpper[0]) valueScore += 2;
         else if(price >= bbMiddle[0]) valueScore += 1;
        }
     }

   // Factor 3: ZLSMA distance - price near or beyond ZLSMA
   if(g_zlsma1 > 0)
     {
      symInfo.RefreshRates();
      double price = isBuy ? symInfo.Ask() : symInfo.Bid();
      double atrBuf[];

      if(SafeCopyBuffer(g_hATR, 0, 1, 1, atrBuf) && ArraySize(atrBuf) >= 1 && atrBuf[0] > 0)
        {
         double distFromZLSMA = (price - g_zlsma1) / atrBuf[0];
         if(isBuy)
           {
            // Buy grid: price below ZLSMA is value
            if(distFromZLSMA < -0.5) valueScore += 1;
           }
         else
           {
            // Sell grid: price above ZLSMA is value
            if(distFromZLSMA > 0.5) valueScore += 1;
           }
        }
     }

   return (valueScore >= 2);
  }

//+------------------------------------------------------------------+
//| GRID MANAGEMENT - Trend-Aware with Dynamic Steps                  |
//+------------------------------------------------------------------+
void ManageGrid()
  {
   if(InpGridMode == GRID_DISABLED)
      return;

   symInfo.Name(_Symbol);
   symInfo.RefreshRates();

   double point = symInfo.Point();
   if(point <= 0) return;

   int digits = (int)symInfo.Digits();

   // Track peak equity for grid DD limit
   double equity = accInfo.Equity();
   if(equity > g_gridPeakEquity)
      g_gridPeakEquity = equity;

   //--- Grid DD limit check: close all if grid drawdown exceeds limit
   if(InpGridDDLimit > 0 && CountPositions() > 1)
     {
      double totalProfit = GetTotalProfit();
      if(totalProfit < 0 && MathAbs(totalProfit) >= InpGridDDLimit)
        {
         CloseAllPositions();
         PrintFormat("[GRID_DD] Grid DD limit hit: $%.2f loss >= $%.2f limit", MathAbs(totalProfit), InpGridDDLimit);
         return;
        }
     }

   //--- Grid partial close: close individually profitable positions
   if(InpGridPartialClose && CountPositions() > 1)
     {
      double partialTP = InpGridPartialTP;
      if(partialTP <= 0)
        {
         // Auto: use ATR * 0.5 as partial TP
         double atrP[];
         if(SafeCopyBuffer(g_hATR, 0, 1, 1, atrP) && ArraySize(atrP) >= 1 && atrP[0] > 0)
            partialTP = atrP[0] / point * 0.5;
         else
            partialTP = 100;
        }

      int total = PositionsTotal();
      for(int i = total - 1; i >= 0; i--)
        {
         if(!posInfo.SelectByIndex(i)) continue;
         if(posInfo.Magic() != InpMagicNumber || posInfo.Symbol() != _Symbol) continue;

         double openP  = posInfo.PriceOpen();
         double profPts = 0;

         if(posInfo.PositionType() == POSITION_TYPE_BUY)
            profPts = (symInfo.Bid() - openP) / point;
         else if(posInfo.PositionType() == POSITION_TYPE_SELL)
            profPts = (openP - symInfo.Ask()) / point;

         if(profPts >= partialTP)
           {
            ulong ticket = posInfo.Ticket();
            if(trade.PositionClose(ticket, InpSlippage))
               PrintFormat("[GRID_PARTIAL] Closed ticket %d at +%.0f pts", ticket, profPts);
           }
        }
     }

   // Grid step calculation
   double gridStep = 0;
   if(InpGridStepPoints > 0)
     {
      gridStep = (double)InpGridStepPoints * point;
     }
   else
     {
      double atrBuf[];
      if(SafeCopyBuffer(g_hATR, 0, 1, 1, atrBuf) && ArraySize(atrBuf) >= 1 && atrBuf[0] > 0)
        {
         gridStep = atrBuf[0] * InpGridATRMultStep;
         // Adapt grid step to volatility
         if(InpUseVolatilityAdapt)
           {
            if(g_volRegime == VOL_HIGH)
               gridStep *= 1.3;
            else if(g_volRegime == VOL_EXTREME)
               gridStep *= 1.8;
            else if(g_volRegime == VOL_LOW)
               gridStep *= 0.7;
           }
        }
      else
         gridStep = 200.0 * point;
     }

   if(gridStep <= 0) return;

   //--- Cooldown check helper: minimum bars between grid additions
   datetime currentBarTime = iTime(_Symbol, _Period, 0);

   // BUY GRID
   int buyCount = GetGridLevel(POSITION_TYPE_BUY);
   if(buyCount > 0 && buyCount < InpGridMaxLevels && CountPositions() < InpMaxPositions)
     {
      // Trend awareness: don't add buy grid if trend is strongly against us
      bool trendBlocked = false;
      if(InpGridTrendAware)
        {
         if(g_trendState == TREND_STRONG_DOWN)
           {
            trendBlocked = true;
            // Exception: allow if grid is very deep (level >= 5) and structure is turning
            if(buyCount >= 5 && g_structureDir >= 0)
               trendBlocked = false;
           }
        }

      // Cooldown: check if enough bars have passed since last grid entry
      bool cooldownOk = true;
      if(InpGridCooldownBars > 0 && g_gridLastBuyTime > 0)
        {
         int barsSinceLastGrid = iBarShift(_Symbol, _Period, g_gridLastBuyTime);
         if(barsSinceLastGrid < InpGridCooldownBars)
            cooldownOk = false;
        }

      if(!trendBlocked && cooldownOk)
        {
         double lastPrice    = GetLastEntryPrice(POSITION_TYPE_BUY);
         double currentPrice = symInfo.Ask();

         // Dynamic step widening: deeper levels get wider steps
         double effectiveStep = gridStep;
         if(InpGridDynamicStep && buyCount > 1)
            effectiveStep = gridStep * (1.0 + (buyCount - 1) * InpGridStepScale);

         if(lastPrice > 0 && currentPrice > 0 && (lastPrice - currentPrice) >= effectiveStep)
           {
            if(IsGridValueZone(true))
              {
               double gridLots = CalcGridLots(buyCount);
               double sl       = CalcStopLoss(true);
               double tp       = 0;

               if(InpGridTPPoints > 0)
                  tp = currentPrice + InpGridTPPoints * point;

               currentPrice = NormalizeDouble(currentPrice, digits);
               sl           = NormalizeDouble(sl, digits);
               if(tp > 0) tp = NormalizeDouble(tp, digits);

               // Free margin check for grid buy entry
               double marginNeededBuy = 0;
               if(OrderCalcMargin(ORDER_TYPE_BUY, _Symbol, gridLots, currentPrice, marginNeededBuy) &&
                  accInfo.FreeMargin() < marginNeededBuy)
                 {
                  PrintFormat("[WARN] Grid BUY Lvl %d skipped: need=%.2f, free=%.2f",
                              buyCount + 1, marginNeededBuy, accInfo.FreeMargin());
                 }
               else
                 {
                  string comment = StringFormat("%s_G%d", InpEAComment, buyCount + 1);
                  if(trade.Buy(gridLots, _Symbol, currentPrice, sl, tp, comment))
                    {
                     g_gridLastBuyTime = currentBarTime;
                     PrintFormat("[GRID] BUY Lvl %d | Price: %.5f | Lots: %.3f | AvgEntry: %.5f | Step: %.1f",
                                 buyCount + 1, currentPrice, gridLots, GetAverageEntryPrice(POSITION_TYPE_BUY),
                                 effectiveStep / point);
                    }
                 }
              }
           }
        }
     }

   // SELL GRID
   int sellCount = GetGridLevel(POSITION_TYPE_SELL);
   if(sellCount > 0 && sellCount < InpGridMaxLevels && CountPositions() < InpMaxPositions)
     {
      // Trend awareness: don't add sell grid if trend is strongly against us
      bool trendBlocked = false;
      if(InpGridTrendAware)
        {
         if(g_trendState == TREND_STRONG_UP)
           {
            trendBlocked = true;
            if(sellCount >= 5 && g_structureDir <= 0)
               trendBlocked = false;
           }
        }

      // Cooldown check
      bool cooldownOk = true;
      if(InpGridCooldownBars > 0 && g_gridLastSellTime > 0)
        {
         int barsSinceLastGrid = iBarShift(_Symbol, _Period, g_gridLastSellTime);
         if(barsSinceLastGrid < InpGridCooldownBars)
            cooldownOk = false;
        }

      if(!trendBlocked && cooldownOk)
        {
         double lastPrice    = GetLastEntryPrice(POSITION_TYPE_SELL);
         double currentPrice = symInfo.Bid();

         // Dynamic step widening
         double effectiveStep = gridStep;
         if(InpGridDynamicStep && sellCount > 1)
            effectiveStep = gridStep * (1.0 + (sellCount - 1) * InpGridStepScale);

         if(lastPrice > 0 && currentPrice > 0 && (currentPrice - lastPrice) >= effectiveStep)
           {
            if(IsGridValueZone(false))
              {
               double gridLots = CalcGridLots(sellCount);
               double sl       = CalcStopLoss(false);
               double tp       = 0;

               if(InpGridTPPoints > 0)
                  tp = currentPrice - InpGridTPPoints * point;

               currentPrice = NormalizeDouble(currentPrice, digits);
               sl           = NormalizeDouble(sl, digits);
               if(tp > 0) tp = NormalizeDouble(tp, digits);

               // Free margin check for grid sell entry
               double marginNeededSell = 0;
               if(OrderCalcMargin(ORDER_TYPE_SELL, _Symbol, gridLots, currentPrice, marginNeededSell) &&
                  accInfo.FreeMargin() < marginNeededSell)
                 {
                  PrintFormat("[WARN] Grid SELL Lvl %d skipped: need=%.2f, free=%.2f",
                              sellCount + 1, marginNeededSell, accInfo.FreeMargin());
                 }
               else
                 {
                  string comment = StringFormat("%s_G%d", InpEAComment, sellCount + 1);
                  if(trade.Sell(gridLots, _Symbol, currentPrice, sl, tp, comment))
                    {
                     g_gridLastSellTime = currentBarTime;
                     PrintFormat("[GRID] SELL Lvl %d | Price: %.5f | Lots: %.3f | AvgEntry: %.5f | Step: %.1f",
                                 sellCount + 1, currentPrice, gridLots, GetAverageEntryPrice(POSITION_TYPE_SELL),
                                 effectiveStep / point);
                    }
                 }
              }
           }
        }
     }

   // BASKET CLOSE - Adaptive to volatility and average entry distance
   if(InpGridBasketClose && CountPositions() > 0)
     {
      double totalProfit = GetTotalProfit();
      double basketTP    = InpGridBasketTP;

      if(basketTP <= 0)
        {
         double bal = accInfo.Balance();
         basketTP = (bal > 0) ? bal * 0.005 : 10.0;

         // Scale basket TP with grid levels
         int totalGridLevels = GetGridLevel(POSITION_TYPE_BUY) + GetGridLevel(POSITION_TYPE_SELL);
         if(totalGridLevels > 3)
            basketTP *= (1.0 + (totalGridLevels - 3) * 0.1);

         // Volatility-adaptive basket TP
         if(InpUseVolatilityAdapt)
           {
            if(g_volRegime == VOL_HIGH)
               basketTP *= 1.3;   // Higher TP in high vol (bigger moves possible)
            else if(g_volRegime == VOL_EXTREME)
               basketTP *= 0.7;   // Lower TP in extreme vol (take what you can)
            else if(g_volRegime == VOL_LOW)
               basketTP *= 0.8;   // Lower TP in low vol (smaller moves)
           }

         // Factor in average entry distance: if positions are far from average,
         // reduce TP to close sooner (positions are under water)
         double buyAvg  = GetAverageEntryPrice(POSITION_TYPE_BUY);
         double sellAvg = GetAverageEntryPrice(POSITION_TYPE_SELL);
         if(buyAvg > 0 && GetGridLevel(POSITION_TYPE_BUY) > 2)
           {
            double avgDist = MathAbs(symInfo.Ask() - buyAvg) / point;
            if(avgDist > 50)
               basketTP *= 0.5;  // Reduce TP when grid is deep underwater
           }
         if(sellAvg > 0 && GetGridLevel(POSITION_TYPE_SELL) > 2)
           {
            double avgDist = MathAbs(sellAvg - symInfo.Bid()) / point;
            if(avgDist > 50)
               basketTP *= 0.5;  // Reduce TP when grid is deep underwater
           }
        }

      if(totalProfit >= basketTP)
        {
         CloseAllPositions();
         PrintFormat("[BASKET] TP reached: $%.2f >= $%.2f", totalProfit, basketTP);
        }
     }
  }

//+------------------------------------------------------------------+
//| TRAILING STOP MANAGEMENT                                          |
//+------------------------------------------------------------------+
void ManageTrailingStop()
  {
   if(!InpUseTrailing && InpBreakEvenPoints <= 0)
      return;

   symInfo.Name(_Symbol);
   symInfo.RefreshRates();

   double point  = symInfo.Point();
   int    digits = (int)symInfo.Digits();
   if(point <= 0) return;

   int total = PositionsTotal();
   for(int i = total - 1; i >= 0; i--)
     {
      if(!posInfo.SelectByIndex(i)) continue;
      if(posInfo.Magic() != InpMagicNumber || posInfo.Symbol() != _Symbol) continue;

      double openPrice = posInfo.PriceOpen();
      double currentSL = posInfo.StopLoss();
      double currentTP = posInfo.TakeProfit();
      double spread    = (double)symInfo.Spread() * point;
      ulong  ticket    = posInfo.Ticket();

      if(posInfo.PositionType() == POSITION_TYPE_BUY)
        {
         double bid    = symInfo.Bid();
         double profit = bid - openPrice;
         if(profit <= 0) continue;

         // Break Even
         if(InpBreakEvenPoints > 0 && profit >= InpBreakEvenPoints * point)
           {
            double newSL = NormalizeDouble(openPrice + spread, digits);
            if(currentSL < newSL)
               trade.PositionModify(ticket, newSL, currentTP);
           }

         // Trailing
         if(InpUseTrailing)
           {
            double trailDist = profit * (1.0 - InpTrailingLevel / 100.0);
            if(trailDist > 0)
              {
               double newSL = NormalizeDouble(bid - trailDist, digits);
               if(newSL > currentSL && newSL > openPrice)
                  trade.PositionModify(ticket, newSL, currentTP);
              }
           }
        }
      else if(posInfo.PositionType() == POSITION_TYPE_SELL)
        {
         double ask    = symInfo.Ask();
         double profit = openPrice - ask;
         if(profit <= 0) continue;

         // Break Even
         if(InpBreakEvenPoints > 0 && profit >= InpBreakEvenPoints * point)
           {
            double newSL = NormalizeDouble(openPrice - spread, digits);
            if(currentSL > newSL || currentSL == 0)
               trade.PositionModify(ticket, newSL, currentTP);
           }

         // Trailing
         if(InpUseTrailing)
           {
            double trailDist = profit * (1.0 - InpTrailingLevel / 100.0);
            if(trailDist > 0)
              {
               double newSL = NormalizeDouble(ask + trailDist, digits);
               if((newSL < currentSL || currentSL == 0) && newSL < openPrice)
                  trade.PositionModify(ticket, newSL, currentTP);
              }
           }
        }
     }
  }

//+------------------------------------------------------------------+
//| EQUITY DRAWDOWN CHECK                                             |
//+------------------------------------------------------------------+
void CheckEquityDrawdown()
  {
   if(InpEquityDDLimit <= 0) return;

   double equity  = accInfo.Equity();
   double balance = accInfo.Balance();
   if(balance <= 0) return;

   double ddPercent = ((balance - equity) / balance) * 100.0;
   if(ddPercent >= InpEquityDDLimit)
     {
      CloseAllPositions();
      PrintFormat("[RISK] EQUITY DD: %.2f%% >= %.2f%%", ddPercent, InpEquityDDLimit);
     }
  }

//+------------------------------------------------------------------+
//| DAILY LOSS CHECK                                                  |
//+------------------------------------------------------------------+
void CheckDailyLoss()
  {
   if(InpMaxDailyLoss <= 0) return;

   MqlDateTime dtNow;
   TimeCurrent(dtNow);

   string dayStr = IntegerToString(dtNow.year) + "." +
                   IntegerToString(dtNow.mon, 2, '0') + "." +
                   IntegerToString(dtNow.day, 2, '0');

   datetime startOfDay = StringToTime(dayStr);
   if(startOfDay != g_dailyResetTime)
     {
      g_dailyStartEquity = accInfo.Equity();
      g_dailyResetTime   = startOfDay;
     }

   double dailyLoss = g_dailyStartEquity - accInfo.Equity();
   if(dailyLoss >= InpMaxDailyLoss)
     {
      CloseAllPositions();
      PrintFormat("[RISK] DAILY LOSS: $%.2f >= $%.2f", dailyLoss, InpMaxDailyLoss);
     }
  }

//+------------------------------------------------------------------+
//| TIME FILTER                                                       |
//+------------------------------------------------------------------+
bool IsWithinTradingTime()
  {
   if(!InpUseTimeFilter) return true;

   MqlDateTime dtNow;
   TimeCurrent(dtNow);

   string dateStr = IntegerToString(dtNow.year) + "." +
                    IntegerToString(dtNow.mon, 2, '0') + "." +
                    IntegerToString(dtNow.day, 2, '0') + " ";

   datetime startTime = StringToTime(dateStr + InpTradingStartTime);
   datetime endTime   = StringToTime(dateStr + InpTradingEndTime);
   datetime now       = TimeCurrent();

   return (now >= startTime && now <= endTime);
  }

//+------------------------------------------------------------------+
//| FRIDAY CLOSE                                                      |
//+------------------------------------------------------------------+
void CheckFridayClose()
  {
   if(!InpCloseFriday) return;

   MqlDateTime dtNow;
   TimeCurrent(dtNow);
   if(dtNow.day_of_week != 5) return;

   string dateStr = IntegerToString(dtNow.year) + "." +
                    IntegerToString(dtNow.mon, 2, '0') + "." +
                    IntegerToString(dtNow.day, 2, '0') + " ";

   datetime closeTime = StringToTime(dateStr + InpFridayCloseTime);

   if(TimeCurrent() >= closeTime && CountPositions() > 0)
     {
      CloseAllPositions();
      Print("[TIME] Friday close executed");
     }
  }

//+------------------------------------------------------------------+
//| CLOSE ALL POSITIONS                                               |
//+------------------------------------------------------------------+
void CloseAllPositions()
  {
   int attempt = 0;
   while(attempt < 5)
     {
      bool allClosed = true;
      int total = PositionsTotal();
      for(int i = total - 1; i >= 0; i--)
        {
         if(posInfo.SelectByIndex(i))
           {
            if(posInfo.Magic() == InpMagicNumber && posInfo.Symbol() == _Symbol)
              {
               if(!trade.PositionClose(posInfo.Ticket(), InpSlippage))
                  allClosed = false;
              }
           }
        }
      if(allClosed || CountPositions() == 0) break;
      Sleep(300);
      attempt++;
     }
  }

//| DETECT FILL TYPE                                                  |
//+------------------------------------------------------------------+
ENUM_ORDER_TYPE_FILLING DetectFillType()
  {
   long fillMode = 0;
   SymbolInfoInteger(_Symbol, SYMBOL_FILLING_MODE, fillMode);

   if((fillMode & SYMBOL_FILLING_FOK) == SYMBOL_FILLING_FOK)
      return ORDER_FILLING_FOK;
   if((fillMode & SYMBOL_FILLING_IOC) == SYMBOL_FILLING_IOC)
      return ORDER_FILLING_IOC;
   return ORDER_FILLING_RETURN;
  }

//+------------------------------------------------------------------+
//| ON TICK                                                           |
//+------------------------------------------------------------------+
void OnTick()
  {
   int minBars = InpCEPeriod;
   if(InpZLSMAPeriod > minBars)       minBars = InpZLSMAPeriod;
   if(InpStructureLookback > minBars) minBars = InpStructureLookback;
   int needBars = minBars * 3 + 50;

   int totalBars = Bars(_Symbol, _Period);
   if(totalBars < needBars)
      return;

   // New bar check
   datetime currentBarTime = iTime(_Symbol, _Period, 0);
   if(currentBarTime == g_lastBarTime)
      return;
   g_lastBarTime = currentBarTime;

   symInfo.RefreshRates();

   // Warm-up period: let indicators stabilize proportionally
   // Uses max(CE, ZLSMA) period / 4 to give recursive CE enough history to converge
   if(!g_initialized)
     {
      g_warmupCount++;
      // Run CE calculation multiple times to establish state
      CalcChandelierExit();

      // Proportional warmup: minimum 5 bars, scaled to indicator complexity
      int requiredWarmup = MathMax(5, MathMax(InpCEPeriod, InpZLSMAPeriod) / 4);
      if(g_warmupCount >= requiredWarmup)
        {
         g_initialized = true;
         Print("[INIT] Warmup complete after ", g_warmupCount, " bars (required: ", requiredWarmup, ")");
        }
      return;
     }

   // Calculate signal with full confluence analysis
   g_lastSignal = GetSignal();

   // Check conditions
   if(!InpAllowNewPos)
     {
      if(g_lastSignal != 0)
         Print("[BLOCK] InpAllowNewPos is false — new trades disabled");
      return;
     }
   if(!IsWithinTradingTime())
     {
      if(g_lastSignal != 0)
         Print("[BLOCK] Outside trading hours — trade skipped");
      return;
     }

   // Grid active with positions — let grid handle
   if(InpGridMode != GRID_DISABLED && CountPositions() > 0)
     {
      if(g_lastSignal != 0)
         PrintFormat("[BLOCK] Grid active with %d positions — ManageGrid() handles entries",
                     CountPositions());
      return;
     }

   // No multiple positions
   if(!InpAllowMultiplePos && CountPositions() > 0)
     {
      if(g_lastSignal != 0)
         PrintFormat("[BLOCK] Already have %d position(s) and AllowMultiplePos=false",
                     CountPositions());
      return;
     }

   // Open trade on signal with sufficient confluence
   if(g_lastSignal != 0)
     {
      PrintFormat("[SIGNAL] Direction: %s | Confluence: %d/%d | %s",
                  g_lastSignal == 1 ? "BUY" : "SELL",
                  g_confluenceScore, InpMinConfluence,
                  g_signalDetails);
      OpenTrade(g_lastSignal);
     }
  }
